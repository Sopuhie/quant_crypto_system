import sqlite3
from typing import Any, dict
from config.settings import DB_PATH

class TradeAnalytics:
@staticmethod
def calculate_metrics() -> dict[str, Any]:
"""Compiles execution orders directly from SQLite to determine real-time strategy performance."""
metrics = {
"total_trades": 0,
"winning_trades": 0,
"losing_trades": 0,
"win_rate": 0.0,
"realized_pnl": 0.0,
"profit_factor": 0.0
}

    try:
        conn = sqlite3.connect(DB_PATH)
        conn.row_factory = sqlite3.Row
        cursor = conn.cursor()
        
        # Fetch all completed or filled orders ordered sequentially
        cursor.execute(\"\"\"
            SELECT side, price, quantity, filled_quantity, status 
            FROM trade_orders 
            WHERE status IN ('closed', 'FILLED') 
            ORDER BY id ASC
        \"\"\")
        orders = cursor.fetchall()
        conn.close()
        
        if not orders:
            return metrics
            
        total_wins = 0
        total_losses = 0
        gross_profit = 0.0
        gross_loss = 0.0
        
        # Simple FIFO / Pairwise match simulation to determine discrete performance flags
        position_entry = None
        for order in orders:
            side = order["side"].lower()
            price = float(order["price"] or 0.0)
            qty = float(order["filled_quantity"] or order["quantity"] or 0.0)
            
            if side == "buy":
                position_entry = price
            elif side == "sell" and position_entry is not None:
                trade_pnl = (price - position_entry) * qty
                metrics["total_trades"] += 1
                metrics["realized_pnl"] += trade_pnl
                
                if trade_pnl > 0:
                    total_wins += 1
                    gross_profit += trade_pnl
                else:
                    total_losses += 1
                    gross_loss += abs(trade_pnl)
                    
                position_entry = None  # Reset tracking lock
        
        # Populate performance matrices safely
        if metrics["total_trades"] > 0:
            metrics["winning_trades"] = total_wins
            metrics["losing_trades"] = total_losses
            metrics["win_rate"] = round((total_wins / metrics["total_trades"]) * 100, 2)
            metrics["profit_factor"] = round(gross_profit / gross_loss, 2) if gross_loss > 0 else round(gross_profit, 2)
            metrics["realized_pnl"] = round(metrics["realized_pnl"], 4)
            
    except Exception:
        pass
        
    return metrics
