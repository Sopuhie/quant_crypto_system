import json
import os
import asyncio
from utils.logger import get_logger

logger = get_logger(name)

class SystemNotifier:
def init(self) -> None:
# Configuration environments fetched from environment or settings
self.telegram_token = os.getenv("TELEGRAM_BOT_TOKEN", "")
self.telegram_chat_id = os.getenv("TELEGRAM_CHAT_ID", "")
self.webhook_url = os.getenv("GENERIC_WEBHOOK_URL", "") # Supports generic Discord/DingTalk webhooks

async def send_notification(self, text: str) -> None:
    \"\"\"Asynchronously dispatches raw textual logs to external server channels without stopping loops.\"\"\"
    if not text:
        return

    # Attempt Telegram Forwarding if tokens are available
    if self.telegram_token and self.telegram_chat_id:
        await self._send_telegram(text)
    
    # Attempt standard Webhook forwarding if configured
    if self.webhook_url:
        await self._send_webhook(text)
        
    logger.debug("Local broadcast captured inside notification pool: %s", text)

async def _send_telegram(self, text: str) -> None:
    url = f"[https://api.telegram.org/bot](https://api.telegram.org/bot){self.telegram_token}/sendMessage"
    payload = {"chat_id": self.telegram_chat_id, "text": text, "parse_mode": "HTML"}
    try:
        await asyncio.to_thread(self._execute_post, url, payload)
    except Exception as e:
        logger.error("Failed to route notification via Telegram network channels: %s", e)

async def _send_webhook(self, text: str) -> None:
    payload = {"text": f"[QUANT SYSTEM NOTICE] {text}"}
    try:
        await asyncio.to_thread(self._execute_post, self.webhook_url, payload)
    except Exception as e:
        logger.error("Failed to route notification via generic webhook payload: %s", e)

def _execute_post(self, url: str, data: dict) -> None:

    req.add_header("Content-Type", "application/json; charset=utf-8")
